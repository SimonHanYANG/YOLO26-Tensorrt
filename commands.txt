============================================================
  YOLO26-TensorRT 命令速查（支持 n/s/m 多模型）
============================================================

所有命令都在项目根目录 /root/yolo26-tensorrt 下执行
conda 环境: yolo  (conda run -n yolo python ...)
GPU: cuda:1

============================================================
1. 训练
============================================================

# YOLO26-N (默认)
conda run -n yolo python train.py

# YOLO26-S
conda run -n yolo python train.py --model yolo26s.pt

# YOLO26-M
conda run -n yolo python train.py --model yolo26m.pt

# 自定义参数
conda run -n yolo python train.py --model yolo26s.pt --epochs 300 --imgsz 640

  输出: runs/detect/train_n/ (n模型)
        runs/detect/train_s/ (s模型)
        runs/detect/train_m/ (m模型)

============================================================
2. 验证 mAP
============================================================

conda run -n yolo python val.py --model n
conda run -n yolo python val.py --model s
conda run -n yolo python val.py --model m

  或指定权重:
conda run -n yolo python val.py --weights runs/detect/train_s/weights/best.pt

============================================================
3. 图片可视化（PyTorch 模型）
============================================================

conda run -n yolo python visualize_predictions.py --model n
conda run -n yolo python visualize_predictions.py --model s
conda run -n yolo python visualize_predictions.py --model m

  输出: visualize_results_n/pred/
        visualize_results_s/pred/
        visualize_results_m/pred/

============================================================
4. 导出 TensorRT Engine（FP32 + FP16）
============================================================

conda run -n yolo python export_engine.py --model n
conda run -n yolo python export_engine.py --model s
conda run -n yolo python export_engine.py --model m

  输出: runs/detect/train_n/weights/best.engine        (FP32)
        runs/detect/train_n/weights/best_fp16.engine   (FP16)
        runs/detect/train_s/weights/best.engine
        runs/detect/train_s/weights/best_fp16.engine
        ...

============================================================
5. TensorRT 图片推理 + 测速
============================================================

# N 模型
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_n/weights/best.engine
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_n/weights/best_fp16.engine

# S 模型
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_s/weights/best.engine
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_s/weights/best_fp16.engine

# M 模型
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_m/weights/best.engine
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_m/weights/best_fp16.engine

  输出: trt_vis_best/pred/  或  trt_vis_best_fp16/pred/

============================================================
6. TensorRT 视频推理（检测）
============================================================

# N 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_n/weights/best.engine

# S 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_s/weights/best.engine

# M 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_m/weights/best.engine

  输出: video_test_results/<engine>_detect/

============================================================
7. TensorRT 视频推理（跟踪 + 轨迹）
============================================================

# N 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_n/weights/best.engine --mode track

# S 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_s/weights/best.engine --mode track

# M 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_m/weights/best.engine --mode track

  输出: video_test_results/<engine>_track/

============================================================
8. 检测 + 跟踪 一起跑（2个视频）
============================================================

conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_n/weights/best.engine --mode both
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_s/weights/best.engine --mode both
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_m/weights/best.engine --mode both

============================================================
9. 检测 + 分割 联合推理（TensorRT + ENet）
============================================================

# 依赖分割项目: /root/Segmentation-Model-Zoo-for-Super-Small-Object/
# 输入视频必须是 1920x1200 (先用 OpenCV 预处理)
# 输出 4 种视频: full / det_track / seg / det_seg

# 先准备 1920x1200 视频 (如果原始视频不是 1920x1200)
# 已有: dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4

# N 模型
conda run -n yolo python det_seg_video.py --engine runs/detect/train_n/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4

# S 模型
conda run -n yolo python det_seg_video.py --engine runs/detect/train_s/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4

# M 模型
conda run -n yolo python det_seg_video.py --engine runs/detect/train_m/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4

  输出: det_seg_results/<video>_<engine>_full.mp4
        det_seg_results/<video>_<engine>_det_track.mp4
        det_seg_results/<video>_<engine>_seg.mp4
        det_seg_results/<video>_<engine>_det_seg.mp4

============================================================
脚本参数速查
============================================================

train.py:
  --model     预训练模型 (yolo26n.pt / yolo26s.pt / yolo26m.pt)
  --data      数据集配置 (默认: dataset/.../data.yaml)
  --epochs    训练轮数 (默认: 200)
  --imgsz     输入尺寸 (默认: 640)
  --device    GPU (默认: 1)
  --name      实验名称 (默认: train_<n/s/m>)

val.py:
  --model     模型大小 (n/s/m)
  --weights   自定义权重路径
  --device    GPU (默认: 1)

export_engine.py:
  --model     模型大小 (n/s/m)
  --weights   自定义权重路径
  --device    GPU (默认: 1)

visualize_predictions.py:
  --model     模型大小 (n/s/m)
  --conf      置信度 (默认: 0.25)
  --device    GPU (默认: 1)

tensorrt_visualize_benchmark.py:
  --engine    engine 路径 (必填)
  --warmup    warmup 次数 (默认: 5)
  --conf      置信度 (默认: 0.25)
  --device    GPU (默认: 1)

tensorrt_video_test.py:
  --engine    engine 路径 (必填)
  --video     视频路径 (默认: gray_video.mp4)
  --mode      detect / track / both (默认: detect)
  --conf      置信度 (默认: 0.25)
  --tracker   跟踪器 (默认: bytetrack.yaml)

det_seg_video.py:
  --engine    engine 路径 (必填)
  --video     视频路径 (必填, 必须 1920x1200)
  --seg_model 分割模型目录 (默认: ENet_sperm_ROINAHead_Xiangya_260817)
  --conf      置信度 (默认: 0.25)
  --seg_alpha 分割叠加透明度 (默认: 0.4)
  --device    GPU (默认: cuda:1)

============================================================

conda run -n yolo python export_engine.py --model s

conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_s/weights/best.engine

# 7. 视频跟踪
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_s/weights/best.engine --mode track

# 8. 检测+分割
conda run -n yolo python det_seg_video.py --engine runs/detect/train_s/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4
