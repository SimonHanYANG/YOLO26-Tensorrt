============================================================
  YOLO26-TensorRT 命令速查（支持 26n/26s/26m 多模型）
============================================================

所有命令都在项目根目录 /root/yolo26-tensorrt 下执行
conda 环境: yolo  (conda run -n yolo python ...)
GPU: cuda:1

输出目录约定:
  训练:  runs/detect/train_26n/   runs/detect/train_26s/   runs/detect/train_26m/
  Engine: weights/best.engine (FP32)  weights/best_fp16.engine (FP16)

============================================================
1. 训练
============================================================

# YOLO26-N (默认)
conda run -n yolo python train.py

# YOLO26-S
conda run -n yolo python train.py --model yolo26s.pt

# YOLO26-M
conda run -n yolo python train.py --model yolo26m.pt

# 自定义参数
conda run -n yolo python train.py --model yolo26s.pt --epochs 300 --imgsz 640

============================================================
2. 验证 mAP
============================================================

conda run -n yolo python val.py --model 26n
conda run -n yolo python val.py --model 26s
conda run -n yolo python val.py --model 26m

  或指定权重:
conda run -n yolo python val.py --weights runs/detect/train_26s/weights/best.pt

============================================================
3. 图片可视化（PyTorch 模型）
============================================================

conda run -n yolo python visualize_predictions.py --model 26n
conda run -n yolo python visualize_predictions.py --model 26s
conda run -n yolo python visualize_predictions.py --model 26m

  输出: visualize_results_26n/pred/
        visualize_results_26s/pred/
        visualize_results_26m/pred/

============================================================
4. 导出 TensorRT Engine（FP32 + FP16）
============================================================

conda run -n yolo python export_engine.py --model 26n
conda run -n yolo python export_engine.py --model 26s
conda run -n yolo python export_engine.py --model 26m

  输出: runs/detect/train_26n/weights/best.engine        (FP32)
        runs/detect/train_26n/weights/best_fp16.engine   (FP16)

============================================================
5. TensorRT 图片推理 + 测速
============================================================

# N 模型
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_26n/weights/best.engine

# S 模型
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_26s/weights/best.engine

# M 模型
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_26m/weights/best.engine

============================================================
6. TensorRT 视频推理（检测）
============================================================

# N 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26n/weights/best.engine

# S 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26s/weights/best.engine

# M 模型
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26m/weights/best.engine

============================================================
7. TensorRT 视频推理（跟踪 + 轨迹）
============================================================

conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26n/weights/best.engine --mode track
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26s/weights/best.engine --mode track
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26m/weights/best.engine --mode track

============================================================
8. 检测 + 跟踪 一起跑
============================================================

conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26n/weights/best.engine --mode both
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26s/weights/best.engine --mode both
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26m/weights/best.engine --mode both

============================================================
9. 检测 + 分割 联合推理（TensorRT + ENet）
============================================================

# 输入视频必须是 1920x1200
# 输出 4 种视频: full / det_track / seg / det_seg

# N 模型
conda run -n yolo python det_seg_video.py --engine runs/detect/train_26n/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4

# S 模型
conda run -n yolo python det_seg_video.py --engine runs/detect/train_26s/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4

# M 模型
conda run -n yolo python det_seg_video.py --engine runs/detect/train_26m/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4

============================================================
脚本参数速查
============================================================

train.py:
  --model     预训练模型 (yolo26n.pt / yolo26s.pt / yolo26m.pt)
  --epochs    训练轮数 (默认: 200)
  --name      实验名称 (默认: train_26n / train_26s / train_26m)

val.py:
  --model     模型标识 (26n / 26s / 26m)
  --weights   自定义权重路径

export_engine.py:
  --model     模型标识 (26n / 26s / 26m)
  --weights   自定义权重路径

visualize_predictions.py:
  --model     模型标识 (26n / 26s / 26m)

tensorrt_visualize_benchmark.py:
  --engine    engine 路径 (必填)

tensorrt_video_test.py:
  --engine    engine 路径 (必填)
  --mode      detect / track / both (默认: detect)

det_seg_video.py:
  --engine    engine 路径 (必填)
  --video     视频路径 (必须 1920x1200)

============================================================
完整流水线 (以 S 模型为例)
============================================================

# 1. 训练
conda run -n yolo python train.py --model yolo26s.pt

# 2. 验证
conda run -n yolo python val.py --model 26s

# 3. 可视化
conda run -n yolo python visualize_predictions.py --model 26s

# 4. 导出 TRT
conda run -n yolo python export_engine.py --model 26s

# 5. 图片测速
conda run -n yolo python tensorrt_visualize_benchmark.py --engine runs/detect/train_26s/weights/best.engine

# 6. 视频检测
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26s/weights/best.engine

# 7. 视频跟踪
conda run -n yolo python tensorrt_video_test.py --engine runs/detect/train_26s/weights/best.engine --mode track

# 8. 检测+分割
conda run -n yolo python det_seg_video.py --engine runs/detect/train_26s/weights/best.engine --video dataset/XiangYa-test-videoes/gray_video_1920x1200.mp4
